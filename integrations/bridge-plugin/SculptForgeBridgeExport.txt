// ZBrush's .ZTL files are proprietary, so SculptForge can't read them. This plug-in runs inside ZBrush and
// writes every SubTool of a ZTool, at its highest subdivision level, as a GoZ file next to the .ZTL:
//     Dragon.ZTL  ->  Dragon.sf001_Body.GoZ, Dragon.sf002_Horns.GoZ, ...
// The .GoZ files keep polygroups, polypaint, the mask and UVs (SculptForge: face groups, vertex paint). Then open Dragon.ZTL in SculptForge: it finds the GoZ
// files and opens them as Parts, in the same order and with the same names.
//
// Install: copy this file into ZBrush's ZStartup/ZPlugs64 folder and restart ZBrush (ZBrush compiles it to a
// .zsc), or load it once with ZScript > Load. The buttons are in ZPlugin > SculptForge.
//
// ZBrush, ZScript, ZTool, GoZ, MatCap and other ZBrush feature names are trademarks of Maxon Computer GmbH. SculptForge is not affiliated with or
// endorsed by Maxon.

[VarDef, sfPath, ""]       // the .ZTL path
[VarDef, sfBase, ""]       // the .ZTL path without its extension
[VarDef, sfFile, ""]
[VarDef, sfName, ""]
[VarDef, sfNum, ""]
[VarDef, sfCount, 0]
[VarDef, sfIndex, 0]
[VarDef, sfActive, 0]

// The .ZTL path in sfPath -> sfBase without ".ZTL" (any letter case).
[RoutineDef, sfStripExtension,
	[VarSet, sfBase, sfPath]
	[If, [StrLength, sfPath] > 4,
		[If, [StrFind, ".ZTL", [StrExtract, sfPath, [StrLength, sfPath] - 4, [StrLength, sfPath] - 1]] == 0,
			[VarSet, sfBase, [StrExtract, sfPath, 0, [StrLength, sfPath] - 5]]
		,
			[If, [StrFind, ".ztl", [StrExtract, sfPath, [StrLength, sfPath] - 4, [StrLength, sfPath] - 1]] == 0,
				[VarSet, sfBase, [StrExtract, sfPath, 0, [StrLength, sfPath] - 5]]
			]
		]
	]
]

// Writes every SubTool of the current ZTool as <sfBase>.sf<NNN>_<SubTool>.GoZ at its highest subdivision level.
[RoutineDef, sfWriteSubTools,
	[VarSet, sfActive, [SubToolGetActiveIndex]]
	[VarSet, sfCount, [SubToolGetCount]]
	[VarSet, sfIndex, 0]
	[Loop, sfCount,
		[SubToolSelect, sfIndex]
		// The highest subdivision level carries the sculpted detail.
		[If, [IsEnabled, "Tool:Geometry:SDiv"],
			[ISet, "Tool:Geometry:SDiv", [IGetMax, "Tool:Geometry:SDiv"]]
		]
		// Tool:ItemInfo is the SubTool's name followed by a dot.
		[VarSet, sfName, [IGetTitle, "Tool:ItemInfo"]]
		[VarSet, sfName, [StrExtract, sfName, 0, [StrLength, sfName] - 2]]
		[VarSet, sfNum, [StrMerge, "00", sfIndex + 1]]
		[VarSet, sfNum, [StrExtract, sfNum, [StrLength, sfNum] - 3, [StrLength, sfNum] - 1]]
		[VarSet, sfFile, [StrMerge, sfBase, ".sf", sfNum, "_", sfName, ".GoZ"]]
		[FileNameSetNext, sfFile]
		[IPress, "Tool:Export"]
		[VarInc, sfIndex]
	]
	[SubToolSelect, sfActive]
	[Note, [StrMerge, "SculptForge: wrote ", sfCount, " SubTools next to ", sfBase, ".ZTL\nOpen the .ZTL in SculptForge."], , 4]
]

[ISubPalette, "ZPlugin:SculptForge"]

[IButton, "ZPlugin:SculptForge:Convert ZTL", "Pick a ZTool (.ZTL): it is loaded and each SubTool is written as a GoZ file next to it, so SculptForge can open the .ZTL.",
	[VarSet, sfPath, [FileNameAsk, "ZTL(*.ZTL)", , "ZTool to convert for SculptForge"]]
	[If, [StrLength, sfPath] > 0,
		[FileNameSetNext, sfPath]
		[IPress, "Tool:Load Tool"]
		[RoutineCall, sfStripExtension]
		[RoutineCall, sfWriteSubTools]
	]
, 0, 1]

[IButton, "ZPlugin:SculptForge:Save for SculptForge", "Save the current ZTool as a .ZTL (choose where) and write each SubTool as a GoZ file next to it, so SculptForge can open the .ZTL.",
	[VarSet, sfPath, [FileNameAsk, "ZTL(*.ZTL)", [IGetTitle, "Tool:Current Tool"], "Save ZTool for SculptForge"]]
	[If, [StrLength, sfPath] > 0,
		[FileNameSetNext, sfPath]
		[IPress, "Tool:Save As"]
		[RoutineCall, sfStripExtension]
		[RoutineCall, sfWriteSubTools]
	]
, 0, 1]
